import time, functools

            last_exc = None
            for i in range(max_retries + 1):
                try:
                    return fn(*args, **kwargs)
                except Exception as e:  # BUG: not filtering to `exceptions`
                    last_exc = e

                        time.sleep(delay)
                    if i == max_retries:
                        raise
            raise last_exc
        return wrapper
    return decorator
